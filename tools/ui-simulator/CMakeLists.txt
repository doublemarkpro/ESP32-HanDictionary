cmake_minimum_required(VERSION 3.22)
project(han_ui_sim LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
file(SHA256 "${LV_BUILD_CONF_PATH}" HAN_UI_CONFIG_HASH)
# Also change compiler flags when config changes: some localized MSVC installations
# fail to record /showIncludes header dependencies reliably.
add_compile_definitions(HAN_UI_CONFIG_HASH="${HAN_UI_CONFIG_HASH}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/platform.h" HAN_PLATFORM_HASH)
file(SHA256 "${ROOT}/main/han_dictionary/han_display.h" HAN_DISPLAY_HASH)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/managed_components/lvgl__lvgl" lvgl)
file(GLOB UI_ASSETS "${ROOT}/main/han_dictionary/assets/*.c")
if(NOT IDF_PATH)
  set(IDF_PATH "$ENV{IDF_PATH}")
endif()
set(JSON_DIR "${ROOT}/managed_components/espressif__cjson/cJSON")
if(NOT EXISTS "${JSON_DIR}/cJSON.c")
  message(FATAL_ERROR "Run tools/setup.ps1 or set IDF_PATH before configuring")
endif()
add_executable(han_ui_sim main.cc "${ROOT}/main/han_dictionary/han_display.cc"
 "${ROOT}/main/han_dictionary/content_store.cc" "${JSON_DIR}/cJSON.c" ${UI_ASSETS})
target_include_directories(han_ui_sim PRIVATE . "${ROOT}/main/han_dictionary" "${JSON_DIR}")
target_compile_definitions(han_ui_sim PRIVATE HAN_UI_HOST_SIM=1 _CRT_SECURE_NO_WARNINGS HAN_SOURCE_ROOT="${ROOT}"
    HAN_PLATFORM_HASH="${HAN_PLATFORM_HASH}" HAN_DISPLAY_HASH="${HAN_DISPLAY_HASH}")
target_link_libraries(han_ui_sim PRIVATE lvgl)
if(MSVC)
  target_compile_options(han_ui_sim PRIVATE /utf-8 /Zc:__cplusplus)
endif()
add_executable(han_graphics_smoke graphics_smoke.cc "${ROOT}/assets/graphics/lvgl/han_graphics_small.c")
target_include_directories(han_graphics_smoke PRIVATE "${ROOT}/assets/graphics/lvgl")
target_link_libraries(han_graphics_smoke PRIVATE lvgl)
if(MSVC)
  target_compile_options(han_graphics_smoke PRIVATE /utf-8 /Zc:__cplusplus)
endif()
